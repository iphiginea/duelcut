async function openDiscover(){ el('discover-modal').classList.remove('hidden'); el('discover-content').innerHTML = `
Finding some hidden gems based on your favorites…
`; state.discoverCandidates = []; if(!state.apiKey){ el('discover-content').innerHTML = `
Add your TMDB API key in Settings first.
`; return; } const sorted = [...state.library].sort((a,b)=>b.elo-a.elo); const seeds = sorted.slice(0, Math.min(5, sorted.length)); if(seeds.length === 0){ el('discover-content').innerHTML = `
Log a few movies first so I have something to work from.
`; return; } const candidates = new Map(); for(const seed of seeds){ try{ const res = await fetch(`https://api.themoviedb.org/3/movie/${seed.id}/recommendations?api_key=${encodeURIComponent(state.apiKey)}`); if(res.ok){ const data = await res.json(); (data.results || []).forEach(m => { if(!state.library.find(lm => lm.id === m.id) && !candidates.has(m.id)){ candidates.set(m.id, m); } }); } }catch(e){ /* skip this seed on failure */ } await new Promise(r => setTimeout(r, 120)); } let list = [...candidates.values()].filter(m => (m.vote_count || 0) >= 15); list.sort((a,b) => (a.popularity || 0) - (b.popularity || 0)); list = list.slice(0, 20); if(list.length === 0){ el('discover-content').innerHTML = `
Couldn't turn up any hidden gems this time — try again after logging a few more movies.
`; return; } state.discoverCandidates = list; showDiscoverBatch(); } function showDiscoverBatch(){ const pool = [...state.discoverCandidates]; const chosen = []; while(chosen.length < 3 && pool.length > 0){ const idx = Math.floor(Math.random()*pool.length); chosen.push(pool.splice(idx,1)[0]); } renderDiscoverResults(chosen); } function renderDiscoverResults(list){ const container = el('discover-content'); container.innerHTML = `

Lesser-known picks based on your top-rated movies. Tap one to open its page on The Movie Database.

`; const grid = el('discover-grid'); list.forEach(movie => { const poster = posterUrl(movie.poster_path, 'w185'); const year = (movie.release_date || '').slice(0,4) || '—'; const card = document.createElement('a'); card.className = 'result-card'; card.href = `https://www.themoviedb.org/movie/${movie.id}`; card.target = '_blank'; card.rel = 'noopener noreferrer'; card.innerHTML = ` ${poster ? `` : `
${escapeHtml(movie.title)}
`}
${escapeHtml(movie.title)}
${year}
`; grid.appendChild(card); }); } function closeDiscover(){ el('discover-modal').classList.add('hidden'); }
